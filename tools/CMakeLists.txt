
if(NOT NO_STATIC_LIB)
	set(MODE static)
else()
	set(MODE shared)
endif()

add_executable(data2cstr data2cstr.c)
target_link_libraries(data2cstr rp-utils-yaml-${MODE})
target_include_directories(data2cstr PRIVATE ../src)
install(TARGETS data2cstr)
message("Adding tool data2cstr")
set(progs data2cstr)

if(TARGET rp-utils-yaml-${MODE})
	add_executable(yaml2json yaml2json.c)
	target_link_libraries(yaml2json rp-utils-yaml-${MODE})
	target_include_directories(yaml2json PRIVATE ../src)
	install(TARGETS yaml2json)
	message("Adding tool yaml2json")
	list(APPEND progs yaml2json)
else()
	message("NO tool yaml2json")
endif()

if(TARGET rp-utils-json-c-${MODE})
	add_executable(info2skel info2skel.c)
	target_link_libraries(info2skel rp-utils-json-c-${MODE})
	target_include_directories(info2skel PRIVATE ../src)
	install(TARGETS info2skel)
	message("Adding tool info2skel")
	list(APPEND progs info2skel)
else()
	message("NO tool info2skel")
endif()

find_program(SCDOC scdoc)
if(SCDOC)
	find_program(GZIP gzip)
	unset(mans)
	foreach(prog ${progs})
		set(in ${CMAKE_CURRENT_SOURCE_DIR}/${prog}.1.scd)
		if(GZIP)
			set(out ${CMAKE_CURRENT_BINARY_DIR}/${prog}.1.gz)
			add_custom_command( OUTPUT ${out} COMMAND ${SCDOC} < ${in} | ${GZIP} > ${out} DEPENDS ${in})
		else(GZIP)
			set(out ${CMAKE_CURRENT_BINARY_DIR}/${prog}.1)
			add_custom_command( OUTPUT ${out} COMMAND ${SCDOC} < ${in} > ${out} DEPENDS ${in})
		endif(GZIP)
		list(APPEND mans ${out})
		install(FILES ${out} DESTINATION ${CMAKE_INSTALL_FULL_MANDIR}/man1)
	endforeach(prog)
	add_custom_target(manuals ALL DEPENDS ${mans})
endif(SCDOC)
